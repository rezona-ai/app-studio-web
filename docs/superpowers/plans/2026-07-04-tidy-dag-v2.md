- [ ] **Step 3: 重写 tidyLayout 函数体**

用下面实现替换 `src/lib/card-layout.ts` 中 `tidyLayout` 的函数体（`105-246` 行；函数头注释同步更新为 DAG 语义）：

```ts
// 一键整理(分量化 DAG):把 parentId 父子边 ∪ refIds 引用边构成的 DAG 重排,返回 id→新坐标。
// 用户主动触发,是"系统永不挪动卡片"原则的有意例外。
// 分组=无向连通分量(引用把多棵树连成一体则整体铺放);列=最长路径分层(节点在其全部父的右侧);
// 行=主父生成树 Reingold–Tilford 初排 + 多父居中修正(入边≥2 的节点 y 对齐各父中心均值,列内下推避让);
// 多分量:各分量局部布好包围盒,再货架式从左到右铺、超 MAX_ROW_WIDTH 换行。
// 锚点取所有卡当前 min(x,y) → 整理后仍停在原区域;末了网格吸附 + 取整匹配后端 int。
export function tidyLayout(nodes: TidyNode[], kindOrder?: string[]): Map<string, XY> {
  const result = new Map<string, XY>();
  if (nodes.length === 0) return result;

  const byId = new Map(nodes.map((n) => [n.id, n]));
  // 有效引用:指向集合内、且不自指(与 selectCanvasEdges 一致)
  const valid = (n: TidyNode, p: string | undefined): p is string => !!p && byId.has(p) && p !== n.id;
  const byPos = (a: TidyNode, b: TidyNode) => a.position.y - b.position.y || a.position.x - b.position.x;
  const w = (n: TidyNode) => n.width ?? CARD_W;
  const h = (n: TidyNode) => n.height ?? CARD_H;

  // 入边父列表:parentId 优先在前,refIds 按数组序去重追加(方向均为 被引用者→本节点)
  const parentsOf = new Map<string, string[]>();
  for (const n of nodes) {
    const ps: string[] = [];
    if (valid(n, n.parentId)) ps.push(n.parentId);
    for (const r of n.refIds ?? []) if (valid(n, r) && !ps.includes(r)) ps.push(r);
    if (ps.length > 0) parentsOf.set(n.id, ps);
  }

  // 无向连通分量:货架铺放单元(引用把多棵树连起来时整体成一个分量);分量序按最左上节点确定性可测
  const adj = new Map<string, string[]>();
  const link = (a: string, b: string) => {
    const arr = adj.get(a);
    if (arr) arr.push(b);
    else adj.set(a, [b]);
  };
  for (const n of nodes) for (const p of parentsOf.get(n.id) ?? []) { link(n.id, p); link(p, n.id); }
  const comps: TidyNode[][] = [];
  const compSeen = new Set<string>();
  for (const seed of [...nodes].sort(byPos)) {
    if (compSeen.has(seed.id)) continue;
    const comp: TidyNode[] = [];
    const stack = [seed.id];
    compSeen.add(seed.id);
    while (stack.length > 0) {
      const id = stack.pop()!;
      comp.push(byId.get(id)!);
      for (const m of adj.get(id) ?? []) if (!compSeen.has(m)) { compSeen.add(m); stack.push(m); }
    }
    comps.push(comp);
  }

  // 全环分量定根:分量内无「零入边」节点时,取最左上节点弃其全部入边作根(保证分层可终止、每分量恒有根)
  for (const comp of comps) {
    if (comp.some((n) => !parentsOf.has(n.id))) continue;
    const root = [...comp].sort(byPos)[0];
    parentsOf.delete(root.id);
  }

  // 主父生成树(首条入边):RT 纵向初排用;childrenOf 只挂主父,保证每节点恰进一棵生成树
  const childrenOf = new Map<string, TidyNode[]>();
  for (const n of nodes) {
    const p = parentsOf.get(n.id)?.[0];
    if (!p) continue;
    const arr = childrenOf.get(p);
    if (arr) arr.push(n);
    else childrenOf.set(p, [n]);
  }
  for (const list of childrenOf.values()) list.sort(byPos);

  // 最长路径分层(列):depth = max(全部父 depth)+1;DFS 栈上回边按 0 兜底(环残边不炸栈)
  const depthOf = new Map<string, number>();
  const onStack = new Set<string>();
  const depth = (id: string): number => {
    const memo = depthOf.get(id);
    if (memo != null) return memo;
    if (onStack.has(id)) return 0; // 回边:忽略(环已在定根阶段拆除,此为双保险)
    onStack.add(id);
    let d = 0;
    for (const p of parentsOf.get(id) ?? []) d = Math.max(d, depth(p) + 1);
    onStack.delete(id);
    depthOf.set(id, d);
    return d;
  };
  for (const n of nodes) depth(n.id);

  // 单个分量布局:返回局部坐标(左上角归零)与包围盒
  const layoutComponent = (comp: TidyNode[]): TreeBox => {
    // 1) 列 x:分量内各列真实最大宽累计
    const colMaxW: number[] = [];
    for (const n of comp) {
      const d = depthOf.get(n.id)!;
      colMaxW[d] = Math.max(colMaxW[d] ?? 0, w(n));
    }
    const colX: number[] = [];
    let acc = 0;
    for (let d = 0; d < colMaxW.length; d++) {
      colX[d] = acc;
      acc += (colMaxW[d] ?? CARD_W) + GAP_X;
    }

    // 2) 纵向初排:主父生成树 RT(place 返回子树纵向区间并写局部坐标,父居中于首末子);
    //    x 用分层列(而非递归深度)——多父节点可能比主父树更深
    const localPos = new Map<string, XY>();
    const seenP = new Set<string>();
    const place = (node: TidyNode, yStart: number): { top: number; bottom: number } => {
      if (seenP.has(node.id)) return { top: yStart, bottom: yStart };
      seenP.add(node.id);
      const x = colX[depthOf.get(node.id)!];
      const kids = childrenOf.get(node.id) ?? [];
      if (kids.length === 0) {
        localPos.set(node.id, { x, y: yStart });
        return { top: yStart, bottom: yStart + h(node) };
      }
      let cursor = yStart;
      const bands: { top: number; bottom: number }[] = [];
      for (const kid of kids) {
        const band = place(kid, cursor);
        bands.push(band);
        cursor = band.bottom + GAP_Y; // 下一兄弟接在上一子树底部之下,子树整体不重叠
      }
      const first = bands[0];
      const last = bands[bands.length - 1];
      const center = ((first.top + first.bottom) / 2 + (last.top + last.bottom) / 2) / 2;
      const y = center - h(node) / 2; // 父纵向居中于首末子
      localPos.set(node.id, { x, y });
      return { top: Math.min(y, first.top), bottom: Math.max(y + h(node), last.bottom) };
    };
    const roots = comp.filter((n) => !parentsOf.has(n.id)).sort(byPos);
    let cursor = 0;
    for (const r of roots) {
      const band = place(r, cursor);
      cursor = band.bottom + GAP_Y; // 多根依次堆叠(等效虚拟超根的兄弟序)
    }

    // 3) 多父居中修正:按列升序处理(父列恒先定稿),入边≥2 的节点 y 对齐各父中心均值;
    //    随后列内按 y 序下推避让,保持 GAP_Y 不重叠
    const byCol = new Map<number, TidyNode[]>();
    for (const n of comp) {
      const d = depthOf.get(n.id)!;
      const arr = byCol.get(d);
      if (arr) arr.push(n);
      else byCol.set(d, [n]);
    }
    for (const d of [...byCol.keys()].sort((a, b) => a - b)) {
      const col = byCol.get(d)!;
      for (const n of col) {
        const ps = (parentsOf.get(n.id) ?? []).filter((p) => (depthOf.get(p) ?? 0) < d && localPos.has(p));
        if (ps.length < 2) continue;
        const centerSum = ps.reduce((s, p) => s + localPos.get(p)!.y + h(byId.get(p)!) / 2, 0);
        localPos.set(n.id, { x: localPos.get(n.id)!.x, y: centerSum / ps.length - h(n) / 2 });
      }
      col.sort((a, b) => localPos.get(a.id)!.y - localPos.get(b.id)!.y || byPos(a, b));
      let bottom: number | null = null;
      for (const n of col) {
        const p = localPos.get(n.id)!;
        const y = bottom == null ? p.y : Math.max(p.y, bottom + GAP_Y);
        localPos.set(n.id, { x: p.x, y });
        bottom = y + h(n);
      }
    }

    // 4) 归一:整分量平移到左上角 (0,0),量包围盒;分带 kind 取首根(定根阶段保证恒有根)
    let minY = Infinity;
    for (const p of localPos.values()) if (p.y < minY) minY = p.y;
    let maxX = 0;
    let maxY = 0;
    for (const [id, p] of localPos) {
      const node = byId.get(id)!;
      const ny = p.y - minY;
      localPos.set(id, { x: p.x, y: ny });
      maxX = Math.max(maxX, p.x + w(node));
      maxY = Math.max(maxY, ny + h(node));
    }
    return { localPos, w: maxX, h: maxY, kind: roots[0]?.kind };
  };

  const boxes: TreeBox[] = comps.map(layoutComponent);

  // 5) 货架打包:按类型分组成行带(kindOrder 缺省则退化为单组 = 原连续货架,逐字节兼容)
  let baseX = Infinity;
  let baseY = Infinity;
  for (const n of nodes) {
    if (n.position.x < baseX) baseX = n.position.x;
    if (n.position.y < baseY) baseY = n.position.y;
  }
  const snap = (v: number) => Math.round(v / SNAP_GRID) * SNAP_GRID;

  const groups: TreeBox[][] = kindOrder ? groupBoxesByKind(boxes, kindOrder) : [boxes];

  let cy = baseY;
  for (const group of groups) {
    if (group.length === 0) continue; // 空类型:不推进 cy、不留空行带
    let cx = baseX;
    let rowH = 0;
    for (const box of group) {
      if (cx > baseX && cx + box.w > baseX + MAX_ROW_WIDTH) {
        cy += rowH + COMPONENT_GAP; // 同类型超宽:本行带内换行
        cx = baseX;
        rowH = 0;
      }
      for (const [id, p] of box.localPos) {
        result.set(id, { x: snap(cx + p.x), y: snap(cy + p.y) });
      }
      cx += box.w + COMPONENT_GAP;
      rowH = Math.max(rowH, box.h);
    }
    cy += rowH + COMPONENT_GAP; // 本类型铺完,下一类型起新行带(组间强制换行,故不同类型 y 区间不相交)
  }
  return result;
}
```
